cmake_minimum_required(VERSION 3.25)
project(dao_sdk_consumer LANGUAGES C CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(DaoKernel 1.1 REQUIRED CONFIG)
add_executable(dao_sdk_consumer main.c)
target_link_libraries(dao_sdk_consumer PRIVATE Dao::dao_kernel)
add_executable(dao_sdk_record_roundtrip record_roundtrip.cpp)
target_compile_features(dao_sdk_record_roundtrip PRIVATE cxx_std_20)
target_link_libraries(dao_sdk_record_roundtrip PRIVATE Dao::dao_assembly Dao::dao_disassembly Dao::dao_module_store)
